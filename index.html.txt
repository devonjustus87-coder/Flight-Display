<script>
    function playAirportChime() {
        try {
            var audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            var now = audioCtx.currentTime;
            
            // Chime 1 (High tone)
            var osc1 = audioCtx.createOscillator();
            var gain1 = audioCtx.createGain();
            osc1.type = 'sine';
            osc1.frequency.setValueAtTime(698.46, now); // F5
            gain1.gain.setValueAtTime(0.3, now);
            gain1.gain.exponentialRampToValueAtTime(0.001, now + 1.2);
            osc1.connect(gain1);
            gain1.connect(audioCtx.destination);
            osc1.start(now);
            osc1.stop(now + 1.2);

            // Chime 2 (Low tone)
            var osc2 = audioCtx.createOscillator();
            var gain2 = audioCtx.createGain();
            osc2.type = 'sine';
            osc2.frequency.setValueAtTime(523.25, now + 0.3); // C5
            gain2.gain.setValueAtTime(0.3, now + 0.3);
            gain2.gain.exponentialRampToValueAtTime(0.001, now + 1.8);
            osc2.connect(gain2);
            gain2.connect(audioCtx.destination);
            osc2.start(now + 0.3);
            osc2.stop(now + 1.8);
        } catch(e) {
            // Fallback for older audio contexts
        }
    }

    function triggerSimulatedPass() {
        playAirportChime();
        document.getElementById('line1').innerHTML = "Emirates";
        document.getElementById('line2').innerHTML = "DXB-LAX";
        document.getElementById('line3').innerHTML = "A380-800";
        document.getElementById('line4').innerHTML = "Arriving from";
        document.getElementById('line5').innerHTML = "Dubai Intl";

        setTimeout(function() {
            document.getElementById('line1').innerHTML = "SYSTEM STANDBY";
            document.getElementById('line2').innerHTML = "-- / --";
            document.getElementById('line3').innerHTML = "SCANNING AIRSPACE";
            document.getElementById('line4').innerHTML = "Status";
            document.getElementById('line5').innerHTML = "Waiting for overhead pass...";
        }, 10000);
    }
</script>
